# Error Handling

All API errors follow a consistent JSON structure, making them easy to parse and handle programmatically.

## Error Format

```json
{
  "error": {
    "code": "ERROR_CODE",
    "message": "Human-readable description"
  }
}
```

The `code` field is a machine-readable string you can use in `switch` statements. The `message` field is a human-readable description suitable for logging.

## Status Codes

| Status | Code | When it happens |
|--------|------|-----------------|
| `400` | `VALIDATION_ERROR` | Missing required parameters, invalid values, or malformed request body |
| `401` | `UNAUTHORIZED` | Missing `Authorization` header or invalid API key |
| `403` | `FORBIDDEN` | API key doesn't have the scope required for this endpoint |
| `404` | `NOT_FOUND` | The requested resource (trend, signal, narrative, etc.) doesn't exist |
| `429` | `RATE_LIMIT_EXCEEDED` | You've exceeded the rate limit — wait and retry |
| `500` | `INTERNAL_ERROR` | Unexpected server error — retry or contact support |

## Handling Errors in Your Code

### TypeScript / JavaScript

```typescript
async function fetchTrends() {
  const response = await fetch("https://khabri.shownomore.com/api/v1/trends", {
    headers: { "Authorization": "Bearer " + process.env.KHABRI_API_KEY }
  });

  if (!response.ok) {
    const { error } = await response.json();

    switch (response.status) {
      case 401:
        throw new Error("Invalid API key. Check your credentials.");
      case 429:
        const resetAt = response.headers.get("X-RateLimit-Reset");
        const waitMs = Number(resetAt) * 1000 - Date.now();
        await new Promise(r => setTimeout(r, waitMs));
        return fetchTrends(); // Retry after waiting
      default:
        throw new Error(`API error: ${error.code} - ${error.message}`);
    }
  }

  return response.json();
}
```

### Python

```python
import requests
import time

def fetch_trends():
    response = requests.get(
        "https://khabri.shownomore.com/api/v1/trends",
        headers={"Authorization": f"Bearer {API_KEY}"}
    )

    if response.status_code == 429:
        reset_at = int(response.headers.get("X-RateLimit-Reset", 0))
        wait = max(reset_at - time.time(), 1)
        time.sleep(wait)
        return fetch_trends()

    response.raise_for_status()
    return response.json()
```

### Go

```go
resp, err := http.DefaultClient.Do(req)
if err != nil {
    log.Fatal(err)
}
defer resp.Body.Close()

if resp.StatusCode != http.StatusOK {
    var errResp struct {
        Error struct {
            Code    string `json:"code"`
            Message string `json:"message"`
        } `json:"error"`
    }
    json.NewDecoder(resp.Body).Decode(&errResp)
    log.Fatalf("API error %d: %s - %s",
        resp.StatusCode, errResp.Error.Code, errResp.Error.Message)
}
```

## Retry with Exponential Backoff

For `429` and `500` errors, implement exponential backoff:

```typescript
async function fetchWithRetry(url: string, options: RequestInit, maxRetries = 3) {
  for (let attempt = 0; attempt <= maxRetries; attempt++) {
    const response = await fetch(url, options);

    if (response.ok) return response.json();

    if (response.status === 429 || response.status >= 500) {
      const delay = Math.pow(2, attempt) * 1000; // 1s, 2s, 4s
      await new Promise(r => setTimeout(r, delay));
      continue;
    }

    // Non-retryable error
    const { error } = await response.json();
    throw new Error(`${error.code}: ${error.message}`);
  }

  throw new Error("Max retries exceeded");
}
```

## Pagination

Paginated endpoints accept `page` (1-based) and `page_size` (default 25, max 100) query parameters. You can also use `limit` as an alias for `page_size`.

Responses include a `meta` object:

```json
{
  "data": [...],
  "meta": {
    "total": 142,
    "page": 1,
    "pageSize": 25,
    "hasMore": true
  }
}
```

### Iterating Through All Pages

```python
import requests

def fetch_all_signals():
    all_signals = []
    page = 1

    while True:
        response = requests.get(
            "https://khabri.shownomore.com/api/v1/signals",
            params={"page": page, "page_size": 100},
            headers={"Authorization": f"Bearer {API_KEY}"}
        )
        data = response.json()
        all_signals.extend(data["data"])

        if not data["meta"]["hasMore"]:
            break
        page += 1

    return all_signals
```
